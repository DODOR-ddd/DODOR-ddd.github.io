    程序清单（main @ 0x00 · SIGINT ISR @ 0x20 · SIGUSR1 ISR @ 0x30）
    
; main: 倒计时 5..1，逐个打印，然后空转
00  LDI R1, 5
03  MOV R0, R1
05  STA 0x80      ; mem[0x80] = R0
07  OUT R1        ; 串口打印
09  ADD R0, -1
0B  MOV R1, R0
0D  JNZ 03
0F  JMP 03        ; idle forever

; SIGINT ISR: 打印当前数据格，覆写成 42，RETI 返回被中断现场
20  LDA 0x80  ·  22  OUT R0  ·  24  LDI R0, 42  ·  27  STA 0x80  ·  29  RETI

; SIGUSR1 ISR: R2 = 1（中断到达的标志），RETI
30  LDI R2, 1  ·  33  RETI
